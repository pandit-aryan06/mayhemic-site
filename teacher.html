harvestStudents() { const studentsMap = new Map(); const box = $('attBox'); if (!box) return []; const trs = box.querySelectorAll('tbody tr, table tr'); trs.forEach((tr, idx) => { if (tr.querySelector('th')) return; const tds = tr.querySelectorAll('td'); if (tds.length >= 2) { let roll = '', rawName = ''; const c0 = tds[0].innerText.trim(); const c1 = tds[1].innerText.trim(); if (/^[0-9A-Za-z-]+$/.test(c0) && c0.length <= 6 && c1.length > 1) { roll = c0; rawName = c1; } else { roll = idx + 1; rawName = c0; } if (!rawName || /^(roll|name|student|total|present|absent)$/i.test(rawName)) return; // SURGICAL EXTRACTION: Grab the text directly from the second cell (Name column) const nameCell = tds[1] || tr.querySelector('td:nth-child(2)'); if (nameCell) { rawName = nameCell.innerText.trim(); } // DEEP SANITIZATION: // 1. Remove newline breaks and extra whitespace // 2. Remove avatar initials attached at the front (e.g. "AR", "SP" preceding actual uppercase names) // 3. Remove "Roll", numbers, "PAL", and attendance states let name = rawName .replace(/[\r\n]+/g, ' ') .replace(/roll\s*\d+/gi, '') .replace(/pal/gi, '') .replace(/(present|absent)/gi, '') .trim(); // Strip leading 1-2 letter uppercase initials if they match or precede the first name // e.g. "ARAryan" -> "Aryan", "SPSaanvi" -> "Saanvi" name = name.replace(/^([A-Z]{1,3})([A-Z][a-z])/, '$2'); name = name.replace(/^([A-Z]{1,2})\s+/, ''); const isAbsent = this.detectRowAbsent(tr); let pDays = null, tDays = null; tds.forEach(td => { const m = td.innerText.match(/(\d+)\s*\/\s*(\d+)/); if (m) { pDays = Number(m[1]); tDays = Number(m[2]); } }); const key = name.toLowerCase(); if (name.length > 1) { studentsMap.set(key, { roll, name, presentDays: pDays ?? 24, totalDays: tDays ?? 24, todayStatus: isAbsent ? 'A' : 'P' }); } } }); return Array.from(studentsMap.values()); },
